; TES4MPQuests — Quest Script (attach to a second persistent quest in the CS)
; Handles outbound quest stage detection and inbound stage application.
;
; Communicates with TES4MPMain via shared variables.

scn TES4MPQuests

; Flags set by TES4MPMain
short bPollQuests
short bApplyIncoming
short bSyncPending

; Set by TES4MPMain when a QUEST_STAGE packet arrives
string_var sIncomingQuestId
short      iIncomingStage

; Last-known stages (one short per quest — update list to match quests.json)
short iLast_MQ00
short iLast_MQ01
short iLast_MQ02
short iLast_MQ03
short iLast_MQ04
short iLast_MQ05
short iLast_MQ06
short iLast_MQ07
short iLast_MQ08
short iLast_MQ09
short iLast_MQ10
short iLast_MQ11
short iLast_MQ12
short iLast_MQ13
short iLast_MQ14
short iLast_MQ15

short iLast_FG01
short iLast_FG02
short iLast_FG03

short iLast_MG01
short iLast_MG02
short iLast_MG03

short iLast_TG01
short iLast_TG02
short iLast_TG03

short iLast_DB01
short iLast_DB02
short iLast_DB03

short iCurrent
string_var sQuestId

begin GameMode

    ; ---- Apply incoming QUEST_STAGE from server ----
    if ( bApplyIncoming == 1 )
        set bApplyIncoming to 0

        ; Only apply if server stage is ahead of local stage
        let iCurrent := GetStage sIncomingQuestId
        if ( iIncomingStage > iCurrent )
            SetStage sIncomingQuestId iIncomingStage
        endif
    endif

    ; ---- Apply full QUEST_SYNC from server ----
    if ( bSyncPending == 1 )
        set bSyncPending to 0

        short iCount
        let iCount := TES4MP_GetSyncCount
        short iIdx
        set iIdx to 0
        label 30
            if ( iIdx >= iCount )
                goto 40
            endif
            short bValid
            let bValid := TES4MP_GetSyncEntry iIdx
            if ( bValid == 1 )
                let sIncomingQuestId := TES4MP_GetPacketStr
                let iIncomingStage   := TES4MP_GetPacketInt
                let iCurrent := GetStage sIncomingQuestId
                if ( iIncomingStage > iCurrent )
                    SetStage sIncomingQuestId iIncomingStage
                endif
            endif
            set iIdx to ( iIdx + 1 )
        goto 30
        label 40
    endif

    ; ---- Poll local quest stages and send changes to server ----
    if ( bPollQuests == 0 )
        return
    endif
    set bPollQuests to 0

    ; Macro-style block: check each quest, send if changed
    ; MQ00
    let iCurrent := GetStage MQ00
    if ( iCurrent != iLast_MQ00 )
        set iLast_MQ00 to iCurrent
        TES4MP_SendQuestStage "MQ00" iCurrent
    endif
    ; MQ01
    let iCurrent := GetStage MQ01
    if ( iCurrent != iLast_MQ01 )
        set iLast_MQ01 to iCurrent
        TES4MP_SendQuestStage "MQ01" iCurrent
    endif
    ; MQ02
    let iCurrent := GetStage MQ02
    if ( iCurrent != iLast_MQ02 )
        set iLast_MQ02 to iCurrent
        TES4MP_SendQuestStage "MQ02" iCurrent
    endif
    ; MQ03
    let iCurrent := GetStage MQ03
    if ( iCurrent != iLast_MQ03 )
        set iLast_MQ03 to iCurrent
        TES4MP_SendQuestStage "MQ03" iCurrent
    endif
    ; MQ04
    let iCurrent := GetStage MQ04
    if ( iCurrent != iLast_MQ04 )
        set iLast_MQ04 to iCurrent
        TES4MP_SendQuestStage "MQ04" iCurrent
    endif
    ; MQ05
    let iCurrent := GetStage MQ05
    if ( iCurrent != iLast_MQ05 )
        set iLast_MQ05 to iCurrent
        TES4MP_SendQuestStage "MQ05" iCurrent
    endif
    ; MQ06
    let iCurrent := GetStage MQ06
    if ( iCurrent != iLast_MQ06 )
        set iLast_MQ06 to iCurrent
        TES4MP_SendQuestStage "MQ06" iCurrent
    endif
    ; MQ07
    let iCurrent := GetStage MQ07
    if ( iCurrent != iLast_MQ07 )
        set iLast_MQ07 to iCurrent
        TES4MP_SendQuestStage "MQ07" iCurrent
    endif
    ; MQ08
    let iCurrent := GetStage MQ08
    if ( iCurrent != iLast_MQ08 )
        set iLast_MQ08 to iCurrent
        TES4MP_SendQuestStage "MQ08" iCurrent
    endif
    ; MQ09
    let iCurrent := GetStage MQ09
    if ( iCurrent != iLast_MQ09 )
        set iLast_MQ09 to iCurrent
        TES4MP_SendQuestStage "MQ09" iCurrent
    endif
    ; MQ10
    let iCurrent := GetStage MQ10
    if ( iCurrent != iLast_MQ10 )
        set iLast_MQ10 to iCurrent
        TES4MP_SendQuestStage "MQ10" iCurrent
    endif
    ; MQ11
    let iCurrent := GetStage MQ11
    if ( iCurrent != iLast_MQ11 )
        set iLast_MQ11 to iCurrent
        TES4MP_SendQuestStage "MQ11" iCurrent
    endif
    ; MQ12
    let iCurrent := GetStage MQ12
    if ( iCurrent != iLast_MQ12 )
        set iLast_MQ12 to iCurrent
        TES4MP_SendQuestStage "MQ12" iCurrent
    endif
    ; MQ13
    let iCurrent := GetStage MQ13
    if ( iCurrent != iLast_MQ13 )
        set iLast_MQ13 to iCurrent
        TES4MP_SendQuestStage "MQ13" iCurrent
    endif
    ; MQ14
    let iCurrent := GetStage MQ14
    if ( iCurrent != iLast_MQ14 )
        set iLast_MQ14 to iCurrent
        TES4MP_SendQuestStage "MQ14" iCurrent
    endif
    ; MQ15
    let iCurrent := GetStage MQ15
    if ( iCurrent != iLast_MQ15 )
        set iLast_MQ15 to iCurrent
        TES4MP_SendQuestStage "MQ15" iCurrent
    endif

    ; FG01-FG03
    let iCurrent := GetStage FG01
    if ( iCurrent != iLast_FG01 ) ; set iLast_FG01 to iCurrent ; TES4MP_SendQuestStage "FG01" iCurrent ; endif
    let iCurrent := GetStage FG02
    if ( iCurrent != iLast_FG02 ) ; set iLast_FG02 to iCurrent ; TES4MP_SendQuestStage "FG02" iCurrent ; endif
    let iCurrent := GetStage FG03
    if ( iCurrent != iLast_FG03 ) ; set iLast_FG03 to iCurrent ; TES4MP_SendQuestStage "FG03" iCurrent ; endif

    ; MG01-MG03
    let iCurrent := GetStage MG01
    if ( iCurrent != iLast_MG01 ) ; set iLast_MG01 to iCurrent ; TES4MP_SendQuestStage "MG01" iCurrent ; endif
    let iCurrent := GetStage MG02
    if ( iCurrent != iLast_MG02 ) ; set iLast_MG02 to iCurrent ; TES4MP_SendQuestStage "MG02" iCurrent ; endif
    let iCurrent := GetStage MG03
    if ( iCurrent != iLast_MG03 ) ; set iLast_MG03 to iCurrent ; TES4MP_SendQuestStage "MG03" iCurrent ; endif

    ; TG01-TG03
    let iCurrent := GetStage TG01
    if ( iCurrent != iLast_TG01 ) ; set iLast_TG01 to iCurrent ; TES4MP_SendQuestStage "TG01" iCurrent ; endif
    let iCurrent := GetStage TG02
    if ( iCurrent != iLast_TG02 ) ; set iLast_TG02 to iCurrent ; TES4MP_SendQuestStage "TG02" iCurrent ; endif
    let iCurrent := GetStage TG03
    if ( iCurrent != iLast_TG03 ) ; set iLast_TG03 to iCurrent ; TES4MP_SendQuestStage "TG03" iCurrent ; endif

    ; DB01-DB03
    let iCurrent := GetStage DB01
    if ( iCurrent != iLast_DB01 ) ; set iLast_DB01 to iCurrent ; TES4MP_SendQuestStage "DB01" iCurrent ; endif
    let iCurrent := GetStage DB02
    if ( iCurrent != iLast_DB02 ) ; set iLast_DB02 to iCurrent ; TES4MP_SendQuestStage "DB02" iCurrent ; endif
    let iCurrent := GetStage DB03
    if ( iCurrent != iLast_DB03 ) ; set iLast_DB03 to iCurrent ; TES4MP_SendQuestStage "DB03" iCurrent ; endif

end
